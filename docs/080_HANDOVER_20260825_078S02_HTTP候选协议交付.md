```bash
curl -X POST http://localhost:8000/api/v1/draft \
  -H 'Authorization: Bearer <token>' \
  -F 'file=@hmi.jpg' \
  -F 'message=' \
  -F 'target=normal_trial.holding' \
  -F 'current_structure={"holding_segments":3}' \
  -F 'allowed_fields=[{"field_id":"holding_segment_pressure","label":"各段保压压力","unit":"bar","shape":"array"}]'
```
